# Standalone build of decomp/tools/rom_decompressor (Extract stage). Mirror of the rom_decompressor target in decomp/tools/Makefile, with the
# vendored miniz standing in for system zlib. Nothing is written into decomp/ or lib/: generated shim headers live in ${CMAKE_BINARY_DIR}.
#   cmake -S tools/windows/extract -B <build>/extract/tool -G Ninja [-DCMAKE_C_COMPILER=clang-cl -DCMAKE_CXX_COMPILER=clang-cl]
cmake_minimum_required(VERSION 3.20)
project(tooie_extract LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

get_filename_component(TOOIE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(DECOMP_TOOLS "${TOOIE_ROOT}/decomp/tools")
set(MINIZ_DIR "${TOOIE_ROOT}/lib/N64ModernRuntime/thirdparty/miniz")
set(SHIM_DIR "${CMAKE_BINARY_DIR}/shim")

# miniz.h includes miniz_export.h (normally generated by miniz's own CMake); zlib.h shim maps rom_decompressor's zlib API onto miniz's
# zlib-compatible names (inflateInit2, inflateReset, MAX_WBITS, z_stream, ...).
file(MAKE_DIRECTORY "${SHIM_DIR}")
file(WRITE "${SHIM_DIR}/miniz_export.h" "#pragma once\n#ifndef MINIZ_EXPORT\n#define MINIZ_EXPORT\n#endif\n")
file(WRITE "${SHIM_DIR}/zlib.h" "#pragma once\n#include \"miniz.h\"\n")

# Inflate only: no deflate, no zip archive code.
add_library(miniz_inflate STATIC "${MINIZ_DIR}/miniz.c" "${MINIZ_DIR}/miniz_tinfl.c")
target_include_directories(miniz_inflate PUBLIC "${SHIM_DIR}" "${MINIZ_DIR}")
target_compile_definitions(miniz_inflate PUBLIC MINIZ_NO_DEFLATE_APIS MINIZ_NO_ARCHIVE_APIS)
if(MSVC)
  target_compile_options(miniz_inflate PRIVATE /w)
else()
  target_compile_options(miniz_inflate PRIVATE -w)
endif()

add_library(fmt_format STATIC "${DECOMP_TOOLS}/fmt/src/format.cc")
target_include_directories(fmt_format PUBLIC "${DECOMP_TOOLS}/fmt/include")

add_library(tooie_utils STATIC "${DECOMP_TOOLS}/tooie_utils.cpp")
target_include_directories(tooie_utils PUBLIC "${DECOMP_TOOLS}")
target_include_directories(tooie_utils PRIVATE
  "${DECOMP_TOOLS}/toml11/include" "${DECOMP_TOOLS}/elfio")
target_link_libraries(tooie_utils PUBLIC fmt_format)

add_executable(rom_decompressor "${DECOMP_TOOLS}/rom_decompressor.cpp")
target_link_libraries(rom_decompressor PRIVATE tooie_utils fmt_format miniz_inflate)

# Vendored sources are not ours to fix: silence their warnings.
foreach(t fmt_format tooie_utils rom_decompressor)
  if(MSVC)
    target_compile_options(${t} PRIVATE /w)
  else()
    target_compile_options(${t} PRIVATE -w)
  endif()
endforeach()
if(MSVC)
  target_compile_definitions(rom_decompressor PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_definitions(tooie_utils PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_definitions(fmt_format PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
