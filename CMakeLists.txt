# TooieRecomp — Banjo-Tooie static recompilation (P05+ executable).
cmake_minimum_required(VERSION 3.20)
project(TooieRecomp LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(BK_ROOT "${CMAKE_SOURCE_DIR}/../BanjoRecomp")
set(N64MR "${BK_ROOT}/lib/N64ModernRuntime")
set(RT64_ROOT "${BK_ROOT}/lib/rt64")
set(RECOMP_FRONTEND "${BK_ROOT}/lib/RecompFrontend")

if (NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Generated RECOMP_FUNC symbols are weak only under Clang. GCC emits strong
# definitions, so boot_hw_stubs cannot interpose. The protected Tooie build
# is Clang; a clean GCC configure reaches link and fails.
if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR
        "TooieRecompiled requires Clang (this compiler is '${CMAKE_C_COMPILER_ID}'). "
        "GCC RECOMP_FUNC is not weak, so boot_hw_stubs cannot override generated "
        "functions. Configure with -DCMAKE_C_COMPILER=clang -DCMAKE_CXX_COMPILER=clang++")
endif()

# Must be CMake variables (not only compile defs) before add_subdirectory(rt64).
set(RT64_SDL_WINDOW_VULKAN TRUE)
set(RT64_STATIC TRUE CACHE BOOL "Build RT64 as a static library" FORCE)
set(PLUME_SDL_VULKAN_ENABLED TRUE CACHE BOOL "Enable SDL Vulkan integration" FORCE)
add_compile_definitions("PLUME_SDL_VULKAN_ENABLED")
add_compile_definitions("RT64_SDL_WINDOW_VULKAN")

find_package(SDL2 REQUIRED)

# --- RT64 + N64ModernRuntime + RecompFrontend (from BanjoRecomp tree) ---
add_subdirectory(${RT64_ROOT} ${CMAKE_BINARY_DIR}/rt64)
set(BUILD_SHARED_LIBS OFF)
add_subdirectory(${N64MR} ${CMAKE_BINARY_DIR}/N64ModernRuntime)
target_include_directories(rt64 PRIVATE ${CMAKE_BINARY_DIR}/rt64/src)

set(RECOMP_FRONTEND_N64MODERNRUNTIME_PATH ${N64MR})
set(RECOMP_FRONTEND_RT64_PATH ${RT64_ROOT})

# DXC + shader opts must be set before RecompFrontend (custom commands expand ${DXC}).
set(DXC_COMMON_OPTS "-I${RT64_ROOT}/src")
set(DXC_DXIL_OPTS "-Wno-ignored-attributes")
set(DXC_SPV_OPTS "-spirv" "-fspv-target-env=vulkan1.0" "-fvk-use-dx-layout")
set(DXC_PS_OPTS ${DXC_COMMON_OPTS} "-E" "PSMain" "-T" "ps_6_0" "-D" "DYNAMIC_RENDER_PARAMS")
set(DXC_VS_OPTS ${DXC_COMMON_OPTS} "-E" "VSMain" "-T" "vs_6_0" "-D" "DYNAMIC_RENDER_PARAMS" "-fvk-invert-y")
set(DXC_CS_OPTS ${DXC_COMMON_OPTS} "-E" "CSMain" "-T" "cs_6_0")
set(DXC_GS_OPTS ${DXC_COMMON_OPTS} "-E" "GSMain" "-T" "gs_6_0")
if(CMAKE_SIZEOF_VOID_P EQUAL 8 AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
  set(DXC "LD_LIBRARY_PATH=${RT64_ROOT}/src/contrib/dxc/lib/x64" "${RT64_ROOT}/src/contrib/dxc/bin/x64/dxc-linux")
else()
  set(DXC "LD_LIBRARY_PATH=${RT64_ROOT}/src/contrib/dxc/lib/arm64" "${RT64_ROOT}/src/contrib/dxc/bin/arm64/dxc-linux")
endif()

add_subdirectory(${RECOMP_FRONTEND} ${CMAKE_BINARY_DIR}/RecompFrontend)

# --- RecompiledFuncs (generated by N64Recomp in P04) ---
add_library(RecompiledFuncs STATIC)
file(GLOB FUNC_C_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/RecompiledFuncs/*.c")
file(GLOB FUNC_CXX_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/RecompiledFuncs/*.cpp")
target_sources(RecompiledFuncs PRIVATE ${FUNC_C_SOURCES} ${FUNC_CXX_SOURCES})
target_compile_options(RecompiledFuncs PRIVATE
    -fno-strict-aliasing
    -Wno-unused-variable
    -Wno-implicit-function-declaration
    -falign-functions=16
)
target_include_directories(RecompiledFuncs PRIVATE
    ${CMAKE_SOURCE_DIR}/include
    ${CMAKE_SOURCE_DIR}/RecompiledFuncs
    ${N64MR}/ultramodern/include
    ${N64MR}/librecomp/include
    ${N64MR}/N64Recomp/include
)

# --- Empty patches lib (P05) ---
# These three files are tracked empty scaffolds, not generated ROM output.
# A clean checkout must already contain them; do not synthesize them here.
set(_TOOIE_PATCH_SCAFFOLDS
    patches.c
    patches_bin.c
    recomp_overlays.inl
)
foreach(_scaffold IN LISTS _TOOIE_PATCH_SCAFFOLDS)
    if(NOT EXISTS "${CMAKE_SOURCE_DIR}/RecompiledPatches/${_scaffold}")
        message(FATAL_ERROR
            "Missing tracked RecompiledPatches scaffold: ${_scaffold}. "
            "These empty files are versioned source inputs, not generated output.")
    endif()
endforeach()

add_library(PatchesLib STATIC
    ${CMAKE_SOURCE_DIR}/RecompiledPatches/patches.c
    ${CMAKE_SOURCE_DIR}/RecompiledPatches/patches_bin.c
)
target_include_directories(PatchesLib PRIVATE
    ${CMAKE_SOURCE_DIR}/include
    ${N64MR}/ultramodern/include
    ${N64MR}/librecomp/include
    ${N64MR}/N64Recomp/include
)

# --- Main executable ---
add_executable(TooieRecompiled
    ${CMAKE_SOURCE_DIR}/src/main/launcher_animation.cpp
    ${CMAKE_SOURCE_DIR}/src/main/main.cpp
    ${CMAKE_SOURCE_DIR}/src/main/register_overlays.cpp

    ${CMAKE_SOURCE_DIR}/src/game/recomp_data_api.cpp
    ${CMAKE_SOURCE_DIR}/src/game/rom_decompression.cpp
    ${CMAKE_SOURCE_DIR}/src/game/syscall_handler.cpp
    ${CMAKE_SOURCE_DIR}/src/game/boot_hw_stubs.cpp
    ${CMAKE_SOURCE_DIR}/src/game/transform_hooks.cpp
    ${CMAKE_SOURCE_DIR}/src/game/tooie_diag.cpp
    ${CMAKE_SOURCE_DIR}/src/game/generated/overlay_dispatch.gen.cpp
    ${CMAKE_SOURCE_DIR}/rsp/n_aspMain.cpp
)

target_include_directories(TooieRecompiled PRIVATE
    ${CMAKE_SOURCE_DIR}/include
    ${CMAKE_SOURCE_DIR}/RecompiledFuncs
    ${CMAKE_SOURCE_DIR}/src/game
    ${CMAKE_BINARY_DIR}
    ${N64MR}/N64Recomp/include
    ${N64MR}/thirdparty/sse2neon
    ${N64MR}/thirdparty
    ${N64MR}/thirdparty/xxHash
    ${RT64_ROOT}/src/contrib
    ${RT64_ROOT}/src/contrib/hlslpp/include
    ${RT64_ROOT}/src/contrib/dxc/inc
    ${RT64_ROOT}/src
    ${RT64_ROOT}/src/rhi
    ${RT64_ROOT}/src/render
    ${RECOMP_FRONTEND}/recompinput/include
    ${RECOMP_FRONTEND}/recompui/include
    ${RECOMP_FRONTEND}/recompui/src
    ${RT64_ROOT}/src/contrib/nativefiledialog-extended/src/include
    ${BK_ROOT}/lib/SlotMap
)

# Fix Banjo-relative stb include in main.cpp by also adding Banjo-style path
target_include_directories(TooieRecompiled PRIVATE ${BK_ROOT})

target_compile_options(TooieRecompiled PRIVATE
    -fno-strict-aliasing
    -fms-extensions
    -Wno-unused-parameter
)

if(CMAKE_SIZEOF_VOID_P EQUAL 8 AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
    target_compile_options(TooieRecompiled PRIVATE -march=nehalem)
endif()

find_package(SDL2 REQUIRED)
find_package(Freetype REQUIRED)
find_package(Threads REQUIRED)

target_include_directories(TooieRecompiled PRIVATE ${SDL2_INCLUDE_DIRS} ${FREETYPE_INCLUDE_DIRS})

target_link_libraries(TooieRecompiled PRIVATE
    PatchesLib
    RecompiledFuncs
    recompui
    recompinput
    librecomp
    ultramodern
    rt64
    nfd
    ${FREETYPE_LIBRARIES}
    SDL2::SDL2
    Threads::Threads
    ${CMAKE_DL_LIBS}
    atomic
)

# Keep controller recognition identical to BanjoRecomp. Make the canonical DB
# an explicit build dependency so updates are recopied even when the executable
# itself does not need relinking.
set(TOOIE_CONTROLLER_DB "${CMAKE_CURRENT_BINARY_DIR}/recompcontrollerdb.txt")
add_custom_command(
    OUTPUT "${TOOIE_CONTROLLER_DB}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${BK_ROOT}/recompcontrollerdb.txt"
        "${TOOIE_CONTROLLER_DB}"
    DEPENDS "${BK_ROOT}/recompcontrollerdb.txt"
    VERBATIM
)
add_custom_target(TooieControllerDB DEPENDS "${TOOIE_CONTROLLER_DB}")
add_dependencies(TooieRecompiled TooieControllerDB)

set (DXC_COMMON_OPTS "-I${RT64_ROOT}/src")
