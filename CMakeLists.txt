# TooieRecomp — Banjo-Tooie static recompilation (P05+ executable).
cmake_minimum_required(VERSION 3.20)
project(TooieRecomp LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Standalone: rt64, N64ModernRuntime (+N64Recomp), RecompFrontend, SlotMap and SDL2 (Windows) are vendored as regular
# files under lib/ (provenance, pins and patch chain: vendor/UPSTREAMS.tsv; integrity: tools/vendor_verify.sh).
# No sibling checkout, submodule or download is used. assets/ and recompcontrollerdb.txt live in this repository.
set(TOOIE_LIB "${CMAKE_SOURCE_DIR}/lib")
set(N64MR "${TOOIE_LIB}/N64ModernRuntime")
set(RT64_ROOT "${TOOIE_LIB}/rt64")
set(RECOMP_FRONTEND "${TOOIE_LIB}/RecompFrontend")
if(NOT EXISTS "${RT64_ROOT}/CMakeLists.txt" OR NOT EXISTS "${N64MR}/CMakeLists.txt" OR NOT EXISTS "${RECOMP_FRONTEND}/CMakeLists.txt")
  message(FATAL_ERROR "vendored tree incomplete under ${TOOIE_LIB}: run tools/vendor_verify.sh")
endif()

if (NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Generated RECOMP_FUNC symbols are weak only under Clang. GCC emits strong
# definitions, so boot_hw_stubs cannot interpose. The protected Tooie build
# is Clang; a clean GCC configure reaches link and fails.
if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR
        "TooieRecompiled requires Clang (this compiler is '${CMAKE_C_COMPILER_ID}'). "
        "GCC RECOMP_FUNC is not weak, so boot_hw_stubs cannot override generated "
        "functions. Configure with -DCMAKE_C_COMPILER=clang -DCMAKE_CXX_COMPILER=clang++")
endif()

# Must be CMake variables (not only compile defs) before add_subdirectory(rt64).
set(RT64_SDL_WINDOW_VULKAN TRUE)
set(RT64_STATIC TRUE CACHE BOOL "Build RT64 as a static library" FORCE)
if(NOT WIN32) # BanjoRecomp sets these on Linux only; Windows keeps its D3D12 default.
  set(PLUME_SDL_VULKAN_ENABLED TRUE CACHE BOOL "Enable SDL Vulkan integration" FORCE)
  add_compile_definitions("PLUME_SDL_VULKAN_ENABLED")
  add_compile_definitions("RT64_SDL_WINDOW_VULKAN")
endif()

if(WIN32)
  # Vendored official SDL2-devel 2.30.3 VC contents (lib/SDL2-windows; provenance vendor/UPSTREAMS.tsv). No FetchContent.
  # RecompFrontend reads sdl2_SOURCE_DIR; rt64 (local seam patch) consumes the SDL2::SDL2/SDL2::SDL2main targets below.
  # UNVERIFIED on a native Windows host (no Windows toolchain on the authoring host): see tools/BUILDING.md.
  set(sdl2_SOURCE_DIR "${TOOIE_LIB}/SDL2-windows")
  if(NOT EXISTS "${sdl2_SOURCE_DIR}/lib/x64/SDL2.lib")
    message(FATAL_ERROR "lib/SDL2-windows is incomplete: run tools/vendor_verify.sh")
  endif()
  set(SDL2_INCLUDE_DIRS "${sdl2_SOURCE_DIR}/include")
  add_library(SDL2::SDL2 INTERFACE IMPORTED)
  set_target_properties(SDL2::SDL2 PROPERTIES
    INTERFACE_INCLUDE_DIRECTORIES "${sdl2_SOURCE_DIR}/include"
    INTERFACE_LINK_LIBRARIES "${sdl2_SOURCE_DIR}/lib/x64/SDL2.lib;Winmm.lib")
  add_library(SDL2::SDL2main INTERFACE IMPORTED)
  set_target_properties(SDL2::SDL2main PROPERTIES
    INTERFACE_LINK_LIBRARIES "${sdl2_SOURCE_DIR}/lib/x64/SDL2main.lib")
  add_compile_definitions(NOMINMAX HLSL_CPU __PRFCHWINTRIN_H)
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -D_DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR")
  if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wno-everything /W4")
  endif()
else()
  find_package(SDL2 REQUIRED)
endif()

# --- Retail cart -> decompressed image (in-memory use of the vendored decomp/tools/rom_decompressor) ---
# Same recipe as tools/windows/extract: the decompressor's zlib calls run on the vendored miniz (already linked by librecomp,
# so no second miniz is compiled) through a zlib.h shim that lives only in the build tree and only on this target's include
# path. fmt is the copy N64Recomp already builds into the game (decomp/tools/fmt is the same major version, v11, in the same
# inline namespace, so a second compiled copy would be a duplicate-symbol link error); tooie_utils is private to this library.
set(TOOIE_DECOMP_TOOLS "${CMAKE_SOURCE_DIR}/decomp/tools")
set(TOOIE_CART_SHIM_DIR "${CMAKE_BINARY_DIR}/cart_decompress_shim")
file(MAKE_DIRECTORY "${TOOIE_CART_SHIM_DIR}")
file(WRITE "${TOOIE_CART_SHIM_DIR}/miniz_export.h" "#pragma once\n#ifndef MINIZ_EXPORT\n#define MINIZ_EXPORT\n#endif\n")
file(WRITE "${TOOIE_CART_SHIM_DIR}/zlib.h" "#pragma once\n#include \"miniz.h\"\n")
add_library(tooie_cart_decompress STATIC
    ${TOOIE_DECOMP_TOOLS}/rom_decompressor.cpp
    ${TOOIE_DECOMP_TOOLS}/tooie_utils.cpp
)
target_include_directories(tooie_cart_decompress PUBLIC ${TOOIE_DECOMP_TOOLS})
target_include_directories(tooie_cart_decompress PRIVATE
    ${TOOIE_CART_SHIM_DIR}
    ${N64MR}/thirdparty/miniz
    ${TOOIE_DECOMP_TOOLS}/toml11/include
    ${TOOIE_DECOMP_TOOLS}/elfio
)
target_compile_definitions(tooie_cart_decompress PRIVATE TOOIE_ROM_DECOMPRESSOR_LIB)
target_link_libraries(tooie_cart_decompress PRIVATE miniz fmt)
# Vendored sources are not ours to fix: silence their warnings (as tools/windows/extract does).
if(MSVC)
    target_compile_options(tooie_cart_decompress PRIVATE /w)
    target_compile_definitions(tooie_cart_decompress PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(tooie_cart_decompress PRIVATE -w)
endif()
